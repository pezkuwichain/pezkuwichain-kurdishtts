import time, requests

API = "https://kurdishtts.dks.news"
H = {"Authorization": "Bearer YOUR_API_KEY"}

job = requests.post(f"{API}/api/tts/jobs", headers=H,
                    json={"text": open("article.txt").read(), "dialect": "kmr"}).json()
while job["status"] in ("queued", "running"):
    time.sleep(5)
    job = requests.get(f"{API}/api/tts/jobs/{job['id']}", headers=H).json()

if job["status"] == "done":
    open("article.mp3", "wb").write(requests.get(API + job["audio_url"]).content)
